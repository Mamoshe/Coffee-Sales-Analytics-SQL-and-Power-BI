--  Total number of sales
SELECT
    COUNT(*) AS total_sales
FROM Sales;


--  Count sales by coffee type
SELECT coffee_name, COUNT(*) AS number_of_sales
FROM Sales
GROUP BY coffee_name
ORDER BY number_of_sales DESC;

-- Which coffee types generate the most revenue?

SELECT
    coffee_name,
    SUM(money) AS total_revenue
FROM Sales
GROUP BY coffee_name
ORDER BY total_revenue DESC;

-- How do transaction volume and revenue vary by payment method?
SELECT
    cash_type AS payment_method,
    COUNT(*) AS total_transactions
FROM Sales
GROUP BY cash_type
ORDER BY total_transactions DESC;

-- Average check
SELECT ROUND(AVG(money)
FROM Sales;

--  Peak hours
SELECT
    EXTRACT(HOUR FROM time) AS sale_hour, COUNT(*) AS number_of_sales, SUM(money) AS total_revenue
FROM Sales
GROUP BY EXTRACT(HOUR FROM time)
ORDER BY number_of_sales DESC;
